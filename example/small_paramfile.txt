# This is a parameter file for ADZE.
# The entire line after a '#' will be ignored.

###----Main Parameters----###
MAX_G 8               #Max standardized sample size

DATA_LINES 22          #Number of lines of data

LOCI 5                #Number of loci

NON_DATA_ROWS 1       #Number of rows preceeding data
                     #including at least the locus names

NON_DATA_COLS 5       #Number of classifier columns
                     #at the beginning of each data line

GROUP_BY_COL 5        #The column number by which to
                     #group the data

DATA_FILE small_data.stru       #Name of the datafile

R_OUT small_r           #Name of the allelic richness output file

P_OUT small_p           #Name of the private allelic richness
                     #output file

###----Combination Parameters----###
COMB 1               #Calculate private allelic richness for
                     #combinations of groupings?

K_RANGE 2             #A listing of combinations to calculate

C_OUT small_c           #Name of the private allelic richness of
                     #combinations output file

###----Advanced Options----###
MISSING -9           #Missing data representation

TOLERANCE 0.2        #Filter loci with a grouping having more
                     #than this fraction of missing data.
                     #0.2 drops D9S1825 and D14S1007, and is the
                     #value that produced the result files
                     #distributed alongside this paramfile.

FULL_R 1             #Output allelic richness results for
                     #all loci?

FULL_P 1             #Output private allelic richness results
                     #for all loci?

FULL_C 1             #Output private allelic richness for
                     #combinations results for all loci?

PRINT_PROGRESS 1     #Track calculation progress on screen?

###----Command line arguments----###
# Any keyword above can be overridden on the command line. Each keyword has a
# long option, and the version 1.0 short flags are still accepted:
#
#   -f  DATA_FILE       --data            -r  R_OUT      --out-richness
#   -g  MAX_G           --max-g           -p  P_OUT      --out-private
#   -l  LOCI            --loci            -o  C_OUT      --out-tuples
#   -d  DATA_LINES      --data-lines      -c  COMB       --combinations
#   -nr NON_DATA_ROWS   --non-data-rows   -k  K_RANGE    --tuples-k
#   -nc NON_DATA_COLS   --non-data-cols   -fr FULL_R     --full-richness
#   -s  GROUP_BY_COL    --group-col       -fp FULL_P     --full-private
#   -m  MISSING         --missing         -fc FULL_C     --full-tuples
#   -t  TOLERANCE       --tolerance       -pp PRINT_...  --progress
#
#   -tnocalc  --tolerance-only    apply the filter, then stop
#   -skipchk  --no-check          accepted, but validation is now free
#
# Options with no version 1.0 equivalent: --out-prefix, --stat, --pops,
# --exclude-pops, --tuples, --tsv, --dry-run, --threads, --quiet, --params.
# Run "adze --help" for the current list with descriptions.
#
# None of the four dimensions above is required any more: LOCI, DATA_LINES and
# NON_DATA_COLS are measured from the datafile, GROUP_BY_COL defaults to the
# last label column, and MAX_G defaults to the largest value the surviving loci
# support. They are kept here because this file doubles as a check that version
# 1.0 paramfiles still work.
###----End of file----###
